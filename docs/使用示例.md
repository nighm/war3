```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
基本使用示例
"""

import sys
from pathlib import Path

# 添加项目根目录到Python路径
sys.path.append(str(Path(__file__).parent.parent))

from src.application.services.project_service import ProjectService
from src.infrastructure.repositories.json_map_repository import JsonMapRepository
from src.domain.services.map_analysis_service import DefaultMapAnalysisService

def main():
    """主函数"""
    print("=== 魔兽争霸3地图开发工作室 - 基本使用示例 ===")
    
    # 初始化服务
    repo = JsonMapRepository()
    analysis_service = DefaultMapAnalysisService()
    project_service = ProjectService(repo, analysis_service)
    
    # 创建新项目
    print("\n1. 创建新项目...")
    project = project_service.create_project(
        name="示例RPG地图",
        project_type="rpg",
        project_path="/tmp/example_project",
        config_data={
            "map_name": "英雄传说",
            "map_description": "一个史诗级的RPG地图",
            "map_author": "示例作者",
            "max_players": 8
        }
    )
    
    print(f"项目创建成功: {project.name}")
    print(f"项目ID: {project.id}")
    print(f"项目类型: {project.project_type}")
    
    # 分析项目
    print("\n2. 分析项目...")
    analysis_result = project_service.analyze_project(str(project.id))
    print("分析完成!")
    
    # 列出所有项目
    print("\n3. 列出所有项目...")
    projects = project_service.list_projects()
    print(f"共有 {len(projects)} 个项目:")
    for p in projects:
        print(f"  - {p.name} ({p.project_type})")
    
    print("\n基本使用示例完成!")

if __name__ == "__main__":
    main()
```

```python
    project.update_project_info(

        description="这是一个更新后的描述"
    )
    
    # 保存项目
    project_service.save_project(project)
    
    # 重新加载项目
    updated_project = project_service.open_project(str(project.id))
    print(f"更新后的项目名称: {updated_project.name}")
    print(f"更新后的描述: {updated_project.description}")
    
    # 添加资源
    project.add_resource("models", "/path/to/hero.mdx")
    project.add_resource("textures", "/path/to/hero.blp")
    project.add_resource("sounds", "/path/to/hero.wav")
    
    print(f"模型资源数量: {project.get_resource_count('models')}")
    print(f"纹理资源数量: {project.get_resource_count('textures')}")
    print(f"音效资源数量: {project.get_resource_count('sounds')}")
    
    # 项目验证
    validation_errors = project.validate_project()
    if validation_errors:
        print("项目验证发现问题:")
        for error in validation_errors:
            print(f"  - {error}")
    else:
        print("项目验证通过!")
    
    # 归档项目
    project.archive_project()
    print(f"项目已归档: {project.is_archived}")
    
    # 激活项目
    project.activate_project()
    print(f"项目已激活: {project.is_active}")
    
    return project
```

```python
def performance_monitoring_example():
    """性能监控示例"""
    
    from src.domain.services.performance_monitor import PerformanceMonitor
    import time
    
    print("创建性能监控器...")
    monitor = PerformanceMonitor(monitoring_interval=0.5)
    
    # 添加告警回调
    def alert_handler(alert):
        print(f"\n🚨 性能告警 [{alert.level.upper()}]: {alert.message}")
        print(f"   当前值: {alert.current_value:.2f}")
        print(f"   阈值: {alert.threshold:.2f}")
        print(f"   建议: {alert.recommendation}")
    
    monitor.add_alert_callback(alert_handler)
    
    # 开始监控
    print("开始性能监控...")
    monitor.start_monitoring()
    
    # 模拟一些工作负载
    print("模拟工作负载...")
    for i in range(10):
        print(f"  执行任务 {i+1}/10...")
        # 模拟CPU密集型任务
        _ = sum(range(1000000))
        time.sleep(0.5)
    
    # 停止监控
    print("停止性能监控...")
    monitor.stop_monitoring()
    
    # 获取性能摘要
    summary = monitor.get_performance_summary()
    print(f"\n性能摘要:")
    print(f"  CPU使用率: {summary.get('cpu_percent', 0):.2f}%")
```

```python
def performance_metrics_analysis():
    """性能指标分析示例"""
    
    from src.domain.services.performance_monitor import PerformanceMonitor
    
    # 创建监控器并加载历史数据
    monitor = PerformanceMonitor()
    
    # 尝试加载历史数据
    try:
        monitor.load_metrics("performance_metrics.json")
        print("已加载历史性能数据")
    except FileNotFoundError:
        print("未找到历史性能数据文件")
        return
    
    # 分析性能趋势
    if monitor.metrics:
        print(f"\n性能趋势分析:")
        print(f"总指标数: {len(monitor.metrics)}")
        
        # 计算平均值
        avg_cpu = sum(m.cpu_percent for m in monitor.metrics) / len(monitor.metrics)
        avg_memory = sum(m.memory_percent for m in monitor.metrics) / len(monitor.metrics)
        
        print(f"CPU平均使用率: {avg_cpu:.2f}%")
        print(f"内存平均使用率: {avg_memory:.2f}%")
        
        # 找出峰值
        max_cpu = max(m.cpu_percent for m in monitor.metrics)
        max_memory = max(m.memory_percent for m in monitor.metrics)
        
        print(f"CPU峰值使用率: {max_cpu:.2f}%")
        print(f"内存峰值使用率: {max_memory:.2f}%")
        
        # 分析告警
        if monitor.alerts:
            print(f"\n告警分析:")
            alert_levels = {}
            for alert in monitor.alerts:
                alert_levels[alert.level] = alert_levels.get(alert.level, 0) + 1
            
            for level, count in alert_levels.items():
                print(f"  {level.upper()} 级别告警: {count} 次")
    
    # 清除指标数据
    print("\n清除性能指标...")
    monitor.clear_metrics()
    print("性能指标已清除")
```

```python
def custom_alert_callbacks():
    """自定义告警回调示例"""
    
    from src.domain.services.performance_monitor import PerformanceMonitor
    import time
    import json
    
    class AlertLogger:
        """告警日志记录器"""
        
        def __init__(self, log_file="alerts.log"):
            self.log_file = log_file
            self.alert_count = 0
        
        def log_alert(self, alert):
            """记录告警"""
            self.alert_count += 1
            log_entry = {
                "timestamp": alert.timestamp.isoformat(),
                "level": alert.level,
                "message": alert.message,
                "metric": alert.metric,
                "current_value": alert.current_value,
                "threshold": alert.threshold,
                "recommendation": alert.recommendation
            }
            
            with open(self.log_file, 'a', encoding='utf-8') as f:
                f.write(json.dumps(log_entry, ensure_ascii=False) + '\n')
            
            print(f"告警已记录到 {self.log_file}")
    
    class AlertNotifier:
        """告警通知器"""
        
        def __init__(self):
            self.critical_alerts = []
        
        def notify_alert(self, alert):
            """处理告警通知"""
            if alert.level == "critical":
                self.critical_alerts.append(alert)
                print(f"🚨 严重告警: {alert.message}")
                print(f"   建议立即处理: {alert.recommendation}")
            elif alert.level == "high":
                print(f"⚠️  高级告警: {alert.message}")
            else:
                print(f"ℹ️  一般告警: {alert.message}")
    
    # 创建监控器
    monitor = PerformanceMonitor(monitoring_interval=1.0)
    
    # 添加自定义回调
    logger = AlertLogger()
    notifier = AlertNotifier()
    
    monitor.add_alert_callback(logger.log_alert)
    monitor.add_alert_callback(notifier.notify_alert)
    
    # 开始监控
    print("开始自定义告警监控...")
    monitor.start_monitoring()
    
    # 模拟工作负载
    time.sleep(5)
    
    # 停止监控
    monitor.stop_monitoring()
    
    print(f"监控完成，共记录 {logger.alert_count} 个告警")
    print(f"严重告警数量: {len(notifier.critical_alerts)}")
    
    return monitor, logger, notifier
```
